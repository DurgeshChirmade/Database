as
  select ename,sal,deptno,job,comm
  from emp;

-ending with n

REGEXP_LIKE()
^:first character
.* : set of characters
$ : last char
[] :set
| : or
. : single char



display all names starting with 'a'
select ename
from emp
where regexp_like(ename,'^a.*');

select ename
from emp
where regexp_like(ename,'^m.*');

select ename
from emp
where regexp_like(ename,'^m');

-name ending with n
select ename
from emp
where regexp_like(ename,'.*n$');

-a can come anywhere
select ename
from emp
where regexp_like(ename,'.*a.*');

-starts with a or m or k
select ename
from emp
where regexp_like(ename,'^[amk].*');

^[amk] --- starts with a or m or k

-starts with m,and l at 3rd pos and ends with r
select ename
from emp
where regexp_like(ename,'^m.l.*r$');

-l at 2nd pos ending with n

select ename
from emp
where regexp_like(ename,'.l.*n$');

-starts with 'a' ends with 'n' or 
starts with 'm' ends with 'r' or
starts with 's' ends with 'h'

select ename
from emp
where regexp_like(ename,'^a.*n$|^m.*r$|^s.*h$');
