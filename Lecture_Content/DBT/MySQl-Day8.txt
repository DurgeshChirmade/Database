begin
  declare p int;
  set p=isperfect(n);
  if p=1
  then 
    select "Yes";
  else
    select "No";
  end if;
end //

delimiter //
create function isperfect(num int) returns int
begin
  declare i,sum int;
  set sum =0;
  set i =1;
  while i<num
  do
    if mod(num,i)=0
    then
      set sum=sum + i;
    end if;
    set i=i+1;
  end while;
  if sum=num
  then
    return 1;
  else 
    return 0;
  end if;
end //
delimiter ;

create procedure job_send(in mjob varchar(15))
begin
   declare p decimal(7,2);
   set p=find_max(mjob);
   select p;
end //

delimiter //
drop function if exists find_max;
create function find_max(m varchar(15)) returns decimal(7,2)
begin
   declare x decimal(7,2);
   select max(sal) into x
   from emp
   where job=m;
   return x;
end //

CREATE TRIGGER add_item
after INSERT
ON order_mst
for each row
begin
  declare a int;
  set a=new.qty*new.price;

  insert into order_log
  values(new.oid,curdate(),a);
end //
