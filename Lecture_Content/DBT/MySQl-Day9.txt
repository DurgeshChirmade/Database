select empno,ename,avg(sal)
from emp;

over()
-------
select empno,ename,deptno,sal,avg(sal) over(partition by deptno) 'avg sal',
max(sal) over(partition by deptno) 'max sal'
from emp;

row_number()
-------------
select ename,deptno,sal,row_number() over()'row number'
from emp;

select ename,deptno,sal,job,row_number() over(partition by job)'row number'
from emp;

select ename,deptno,sal,job,row_number() over(partition by job)'row number'
from emp
where job='manager';

rank()
------
select ename,deptno,sal,job,rank() over(partition by job order by sal) rnk
from emp;

select ename,deptno,sal,job,rank() over(partition by job order by sal desc) rnk
from emp;

dens_rank()
--------
select ename,deptno,sal,job,rank() over(partition by job order by sal) rnk,dense_rank() over(partition by job order by sal) drnk
from emp;

lag()
-----
select ename,deptno,sal,job,lag(sal) over(partition by job) 'prev sal'
from emp;

lead()
------
select ename,deptno,sal,job,lead(sal) over(partition by job) 'next sal'
from emp;
