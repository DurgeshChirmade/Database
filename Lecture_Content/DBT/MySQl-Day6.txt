right outer join
--------------
select e.empno,e.ename,e.deptno,d.dname
from dept d
right join emp e
on e.deptno=d.deptno;
 


select e.empno,e.ename,e.deptno,e.job,d.dname
from dept d
right join emp e
on e.deptno=d.deptno
where e.job='salesman';


full join
------------
select e.empno,e.ename,d.deptno,d.dname
from dept d
left join emp e
on e.deptno=d.deptno
union
select e.empno,e.ename,e.deptno,d.dname
from dept d
right join emp e
on e.deptno=d.deptno;
