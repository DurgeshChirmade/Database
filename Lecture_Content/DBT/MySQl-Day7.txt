views in RDBMS
=================

create view x
as
  select ename,sal,comm
  from emp;

insert into x
values('abcd',30000,1000);

delete from x
where ename='abcd';

============
create view dept20
as
  select * 
  from emp
  where deptno=20;

 insert into dept20(ename,deptno)
 values('tttt',20);

 insert into dept20(ename,deptno)
 values('RRRR',30);

create view dept20_new
as
  select * 
  from emp
  where deptno=20
  with check option;

 insert into dept20(ename,deptno)
 values('RRRR',30);


create view z
as
select e.ename, e.deptno, d.dname
from emp e
join dept d
on e.deptno=d.deptno;
