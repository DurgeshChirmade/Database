create table dept
(dno int,
 dname varchar(15),
 constraint pk_dno primary key(dno)
);
insert into dept
values(100,'R&D'),
       (200,'ACC'),
       (300,'Sales');

create table emp
(eno int,
 ename varchar(20) not null,
 deptno int,
 constraint pk_eno primary key(eno),
 constraint fk_no foreign key(deptno) references dept(dno)
);
 
insert into emp
values(1,'Sujay',200);


group by
--------

select count(*)
from emp;

select deptno,count(*)
from emp
group by deptno;
