delimiter //
create procedure second_cursor()
begin
	declare a int;
	declare b varchar(15);
	declare c varchar(15);
	declare d int;
	declare e date;
	declare f decimal(7,2);
	declare g decimal(7,2);
	declare h int;
	declare flag int default 0;
	declare cur cursor for
		select e.empno,e.ename,e.job,e.mgr,e.hiredate,e.sal,e.comm,e.deptno
		from emp e;
	declare continue handler for NOT FOUND
		set flag=1;
	open cur;
	print:loop
		fetch cur into a,b,c,d,e,f,g,h;

		if flag=1
		then
			leave print;
		end if;

		if c='clerk'
		then
			set f=f+(f*10)/100;
		elseif c='manager'
		then 
			set f=f+(f*5)/100;
		elseif c='analyst'
		then
			set f=f+(f*7)/100;
		end if;

		select a,b,c,d,e,f,g,h;
	end loop print;
	close cur;
end //
delimiter;

call second_cursor;
