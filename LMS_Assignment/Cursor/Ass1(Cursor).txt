delimiter //
create procedure first_cursor()
begin
	declare a int;
	declare b varchar(15);
	declare c varchar(15);
	declare d int;
	declare e date;
	declare f decimal(7,2);
	declare g decimal(7,2);
	declare h int;
	declare i varchar(15);
	declare flag int default 0;
	declare cur cursor for
		select e.empno,e.ename,e.job,e.mgr,e.hiredate,e.sal,e.comm,e.deptno,d.dname
		from emp e
		join dept d
		on e.deptno=d.deptno
		where e.deptno=20;
	declare continue handler for NOT FOUND
		set flag=1;
	open cur;
	print:loop
		fetch cur into a,b,c,d,e,f,g,h,i;

		if flag=1
		then
			leave print;
		end if;
		select a,b,c,d,e,f,g,h,i;
	end loop print;
	close cur;
end //
delimiter;

call first_cursor;
