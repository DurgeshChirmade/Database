create table product_details(pid int,amt float);

delimiter //
create trigger first_trigger
after update 
on product
for each row
begin
	if new.price <> old.price
	then
		INSERT INTO product_details (pid, amt)
    	VALUES (NEW.pid, NEW.price * NEW.qty);
	end if;
end //
delimiter ;
