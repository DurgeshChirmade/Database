create function total_sal(s decimal(7,2),c decimal(7,2)) 
returns decimal(7,2)
begin
	declare tot_sal decimal(7,2);

	if c is null
	then
		set tot_sal=s;
	else
		set tot_sal=s + c;
	end if;

	return tot_sal;

end //

	declare g decimal(7,2);

		select e.empno,e.ename,e.job,e.mgr,e.hiredate,e.sal,e.comm,e.deptno,d.dname
		from emp e

		on e.deptno=d.deptno

		then
			leave print;

create table product_details(pid int,amt float);

delimiter //
create trigger first_trigger
after update 
on product
for each row
begin
	if new.price <> old.price
	then
		update product_details
		set amt = new.price * (select qty from product where pid=new.pid)
		where pid=new.pid;
	end if;
end //
delimiter ;

delimiter //
create trigger first_trigger
after update 
on product
for each row
begin
	if new.price <> old.price
	then
		INSERT INTO product_details (pid, amt)
    	VALUES (NEW.pid, NEW.price * NEW.qty);

end //
