set global log_bin_trust_function_creators=1;

delimiter //
create function total_sal(s decimal(7,2),c decimal(7,2)) 
returns decimal(7,2)
begin
	declare tot_sal decimal(7,2);

	if c is null
	then
		set tot_sal=s;
	else
		set tot_sal=s + c;
	end if;

	return tot_sal;

end //
delimiter ;

create function tota_sal(s decimal(7,2),c decimal(7,2)) 
returns decimal(7,2)
begin
	return s+ifnull(c,0);
